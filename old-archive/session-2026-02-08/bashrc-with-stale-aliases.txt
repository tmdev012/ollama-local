#force_color_prompt=yes

	# We have color support; assume it's compliant with Ecma-48
	# (ISO/IEC-6429). (Lack of such support is extremely rare, and such
	# a case would tend to support setf rather than setaf.)
	color_prompt=yes
    else
	color_prompt=
    fi
fi

if [ "$color_prompt" = yes ]; then
    PS1='${debian_chroot:+($debian_chroot)}\[\033[01;32m\]\u@\h\[\033[00m\]:\[\033[01;34m\]\w\[\033[00m\]\$ '
else
    PS1='${debian_chroot:+($debian_chroot)}\u@\h:\w\$ '
fi

# ============================================
# GITPUSH - One command add+commit+push
# ============================================
gitpush() {
    local msg="${1:-Auto-commit $(date +%Y-%m-%d\ %H:%M)}"

    echo "📦 Staging all changes..."
    git add -A

    echo "📝 Committing: $msg"
    git commit -m "$msg

Co-Authored-By: Claude Opus 4.5 <noreply@anthropic.com>"

    echo "🚀 Pushing to origin..."
    git push

    echo "✅ Done!"
}

# Quick commit history from SQLite
git-history() {
    python3 << 'PYEOF'
import sqlite3
import os
db = os.path.expanduser('~/ollama-local/db/history.db')
conn = sqlite3.connect(db)
c = conn.cursor()
c.execute('SELECT hash, message, branch, files_changed, timestamp FROM commits ORDER BY id DESC LIMIT 20')
rows = c.fetchall()
if rows:
    print(f"{'Hash':<10} {'Message':<40} {'Branch':<10} {'Files':<6} {'Time'}")
    print("-" * 85)
    for r in rows:
        msg = (r[1] or '')[:38]
        print(f"{r[0]:<10} {msg:<40} {r[2]:<10} {r[3] or 0:<6} {r[4]}")
else:
    print("No commits tracked yet. Use: smartpush")
PYEOF
}

# View commit history with categories
git-history() {
    python3 << 'PYEOF'
import sqlite3
import os
db = os.path.expanduser('~/ollama-local/db/history.db')
conn = sqlite3.connect(db)
c = conn.cursor()
try:
    c.execute('''SELECT hash, message, version_tag, issue_number, branch, 
                        files_changed, lines_added, lines_deleted, categories, timestamp 
                 FROM commits ORDER BY id DESC LIMIT 15''')
    rows = c.fetchall()
    if rows:
        print(f"{'Hash':<9} {'Ver':<8} {'Issue':<6} {'Cat':<12} {'+/-':<10} {'Message':<35}")
        print("─" * 90)
        for r in rows:
            ver = r[2] or "-"
            issue = f"#{r[3]}" if r[3] else "-"
            cat = (r[8] or "other")[:10]
            changes = f"+{r[6] or 0}/-{r[7] or 0}"
            msg = (r[1] or '')[:33]
            print(f"{r[0]:<9} {ver:<8} {issue:<6} {cat:<12} {changes:<10} {msg}")
    else:
        print("No commits tracked. Use: smartpush")
except:
    print("Run smartpush first to initialize tracking")
PYEOF
}

# View by issue
git-issue() {
    local issue="$1"
    [ -z "$issue" ] && { echo "Usage: git-issue <number>"; return 1; }
    python3 << PYEOF
import sqlite3, os
conn = sqlite3.connect(os.path.expanduser('~/ollama-local/db/history.db'))
c = conn.cursor()
c.execute("SELECT hash, message, version_tag, timestamp FROM commits WHERE issue_number=? ORDER BY id DESC", ('$issue',))
rows = c.fetchall()
print(f"Commits for issue #$issue:")
print("─" * 60)
for r in rows:
    print(f"  {r[0]} | {r[2] or '-':<8} | {r[1][:40]}")
    print(f"           {r[3]}")
PYEOF
}

# View by version
git-version() {
    python3 << 'PYEOF'
import sqlite3, os
conn = sqlite3.connect(os.path.expanduser('~/ollama-local/db/history.db'))
c = conn.cursor()
c.execute("SELECT DISTINCT version_tag, hash, message, timestamp FROM commits WHERE version_tag IS NOT NULL ORDER BY id DESC LIMIT 10")
rows = c.fetchall()
print("Version Tags:")
print("─" * 60)
for r in rows:
    print(f"  {r[0]:<10} {r[1]} | {r[2][:35]}")
PYEOF
}
